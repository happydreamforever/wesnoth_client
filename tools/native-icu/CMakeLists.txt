cmake_minimum_required(VERSION 3.22)
project(wesnoth_android_icu LANGUAGES C CXX ASM)

# A deliberately bounded Android recipe. The release source lists and packaged
# little-endian data are checked by build_android_icu.py before configuring.
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "x86_64" OR
   NOT ANDROID_PLATFORM_LEVEL STREQUAL "23" OR NOT ANDROID_STL STREQUAL "c++_shared")
    message(FATAL_ERROR "Use the selected Android x86_64/API23/shared-C++ toolchain.")
endif()
if(NOT IS_DIRECTORY "${ICU_SOURCE}/source/common" OR NOT EXISTS "${ICU_DATA_ASSEMBLY}")
    message(FATAL_ERROR "The verified owned ICU sources and data assembly are required.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH ON)

add_library(icudata STATIC "${ICU_DATA_ASSEMBLY}")
set_target_properties(icudata PROPERTIES LINKER_LANGUAGE C)
foreach(component common i18n)
    file(STRINGS "${ICU_SOURCE}/source/${component}/sources.txt" source_names)
    set(sources)
    foreach(name IN LISTS source_names)
        if(NOT name MATCHES "^[a-zA-Z0-9_-]+\\.cpp$")
            message(FATAL_ERROR "Unexpected ICU source-list entry: ${name}")
        endif()
        list(APPEND sources "${ICU_SOURCE}/source/${component}/${name}")
    endforeach()
    if(component STREQUAL "common")
        set(target icuuc)
        set(implementation U_COMMON_IMPLEMENTATION)
    else()
        set(target icui18n)
        set(implementation U_I18N_IMPLEMENTATION)
    endif()
    add_library(${target} STATIC ${sources})
    target_compile_definitions(${target} PUBLIC U_STATIC_IMPLEMENTATION PRIVATE ${implementation})
    target_compile_options(${target} PRIVATE -O2)
    target_include_directories(${target} PUBLIC "${ICU_SOURCE}/source/common" "${ICU_SOURCE}/source/i18n")
    install(DIRECTORY "${ICU_SOURCE}/source/${component}/unicode/"
        DESTINATION include/unicode FILES_MATCHING PATTERN "*.h")
endforeach()
target_link_libraries(icuuc PUBLIC icudata dl m)
target_link_libraries(icui18n PUBLIC icuuc)
install(TARGETS icudata icuuc icui18n ARCHIVE DESTINATION lib)
install(FILES "${ICU_SOURCE}/LICENSE" DESTINATION share/licenses/wesnoth-icu)

add_executable(wesnoth-icu-check probe.cpp)
target_compile_options(wesnoth-icu-check PRIVATE -Wall -Wextra -Werror -fexceptions)
target_link_options(wesnoth-icu-check PRIVATE -pie "-Wl,-Map,${CMAKE_BINARY_DIR}/probe.map")
target_link_libraries(wesnoth-icu-check PRIVATE icui18n)
set_target_properties(wesnoth-icu-check PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS wesnoth-icu-check RUNTIME DESTINATION bin)
