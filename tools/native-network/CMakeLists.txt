cmake_minimum_required(VERSION 3.22)
project(wesnoth_android_network LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "x86_64" OR
   NOT ANDROID_PLATFORM_LEVEL STREQUAL "23" OR NOT ANDROID_STL STREQUAL "c++_shared")
    message(FATAL_ERROR "Use Android x86_64/API23/shared-C++ toolchain.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH ON)
set(CMAKE_INSTALL_MESSAGE NEVER)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/build-plan.json" plan)
# Use the same pinned file lists as the verifier. Windows wildcard matching of
# *.h would also install OpenSSL's VMS-only __DECC_INCLUDE_*.H source files.
function(install_public_headers component source)
    string(JSON count LENGTH "${plan}" public_header_paths ${component})
    math(EXPR last "${count}-1")
    set(headers)
    foreach(index RANGE ${last})
        string(JSON name GET "${plan}" public_header_paths ${component} ${index})
        list(APPEND headers "${source}/include/${name}")
    endforeach()
    install(FILES ${headers} DESTINATION "include/${component}")
endfunction()

foreach(component crypto ssl)
    set(sources)
    string(JSON count LENGTH "${plan}" openssl_graph lib${component} sources)
    math(EXPR last "${count}-1")
    foreach(index RANGE ${last})
        string(JSON name GET "${plan}" openssl_graph lib${component} sources ${index})
        list(APPEND sources "${OPENSSL_SOURCE}/${name}")
    endforeach()
    add_library(${component} STATIC ${sources})
    target_compile_definitions(${component} PRIVATE OPENSSL_BUILDING_OPENSSL OPENSSL_USE_NODELETE OPENSSL_PIC NDEBUG
        "OPENSSLDIR=\"/wesnoth-unused-ssl\"" "ENGINESDIR=\"/wesnoth-unused-engines\"" "MODULESDIR=\"/wesnoth-unused-modules\"")
    target_compile_options(${component} PRIVATE -O2 -pthread)
    foreach(field includes defines)
        string(JSON count LENGTH "${plan}" openssl_graph lib${component} ${field})
        if(count GREATER 0)
            math(EXPR last "${count}-1")
            foreach(index RANGE ${last})
                string(JSON name GET "${plan}" openssl_graph lib${component} ${field} ${index})
                if(field STREQUAL "includes")
                    target_include_directories(${component} PRIVATE "${OPENSSL_SOURCE}/${name}")
                else()
                    target_compile_definitions(${component} PRIVATE "${name}")
                endif()
            endforeach()
        endif()
    endforeach()
    target_include_directories(${component} PUBLIC "${OPENSSL_SOURCE}/include")
endforeach()
target_link_libraries(ssl PUBLIC crypto)
target_link_libraries(crypto PUBLIC dl)
install(TARGETS crypto ssl ARCHIVE DESTINATION lib)
install_public_headers(openssl "${OPENSSL_SOURCE}")
install(FILES "${OPENSSL_SOURCE}/LICENSE.txt" DESTINATION share/licenses/wesnoth-network RENAME openssl-LICENSE.txt)
