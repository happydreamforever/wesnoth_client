cmake_minimum_required(VERSION 3.22)
project(wesnoth_android_boost LANGUAGES C CXX ASM)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "x86_64" OR
   NOT ANDROID_PLATFORM_LEVEL STREQUAL "23" OR NOT ANDROID_STL STREQUAL "c++_shared")
    message(FATAL_ERROR "Use the selected Android x86_64/API23/shared-C++ toolchain.")
endif()
if(NOT EXISTS "${BOOST_SOURCE}/boost/version.hpp" OR NOT EXISTS "${ICU_PREFIX}/lib/libicui18n.a")
    message(FATAL_ERROR "Verified owned Boost source and prior ICU prefix are required.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH ON)
# Keep a 16,010-header install readable; native failures remain visible.
set(CMAKE_INSTALL_MESSAGE NEVER)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/build-plan.json" plan)

# Pin the upstream source lists, but drive Clang directly rather than build a
# host b2 executable. No discovery of optional host compression/locale libraries.
set(boost_components atomic charconv chrono context coroutine filesystem iostreams locale process program_options random thread)
foreach(component IN LISTS boost_components)
    string(JSON count LENGTH "${plan}" libraries ${component})
    math(EXPR last "${count}-1")
    set(sources)
    foreach(index RANGE ${last})
        string(JSON name GET "${plan}" libraries ${component} ${index})
        list(APPEND sources "${BOOST_SOURCE}/libs/${component}/${name}")
    endforeach()
    add_library(boost_${component} STATIC ${sources})
    string(TOUPPER "${component}" upper)
    target_include_directories(boost_${component} SYSTEM PUBLIC "${BOOST_SOURCE}")
    target_compile_definitions(boost_${component} PUBLIC BOOST_ALL_NO_LIB BOOST_${upper}_STATIC_LINK)
    # These sources define their own implementation macro, as noted upstream.
    if(NOT component STREQUAL "iostreams")
        target_compile_definitions(boost_${component} PRIVATE BOOST_${upper}_SOURCE)
    endif()
    target_compile_options(boost_${component} PRIVATE -O2)
    list(APPEND install_targets boost_${component})
endforeach()

# Match the release's optional x86 address lookup accelerators; selection inside
# the atomic library remains runtime-dispatched by its unchanged lock_pool.cpp.
set_source_files_properties("${BOOST_SOURCE}/libs/atomic/src/find_address_sse2.cpp" PROPERTIES COMPILE_OPTIONS "-msse;-msse2")
set_source_files_properties("${BOOST_SOURCE}/libs/atomic/src/find_address_sse41.cpp" PROPERTIES COMPILE_OPTIONS "-msse;-msse2;-msse3;-mssse3;-msse4.1")
target_include_directories(boost_atomic PRIVATE "${BOOST_SOURCE}/libs/atomic/src")
target_include_directories(boost_filesystem PRIVATE "${BOOST_SOURCE}/libs/filesystem/src")
target_compile_definitions(boost_atomic PRIVATE BOOST_ATOMIC_USE_SSE2 BOOST_ATOMIC_USE_SSE41)
target_compile_definitions(boost_context PUBLIC BOOST_CONTEXT_EXPORT=)
target_compile_definitions(boost_coroutine PRIVATE BOOST_COROUTINES_SOURCE)
target_compile_definitions(boost_thread PRIVATE BOOST_THREAD_BUILD_LIB PUBLIC BOOST_THREAD_USE_LIB)
target_compile_definitions(boost_charconv PUBLIC BOOST_CHARCONV_NO_QUADMATH)

# These are compile/link capability tests against the selected Android sysroot;
# no Android executable is run by CMake. Newer optional filesystem syscalls are
# disabled for the API23 package; POSIX fallbacks remain available.
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES "${BOOST_SOURCE}" "${BOOST_SOURCE}/libs/filesystem/src")
foreach(feature init_priority stat_st_blksize stat_st_mtim stat_st_mtimensec stat_st_mtimespec stat_st_birthtimensec stat_st_birthtimespec fdopendir_nofollow dirent_d_type posix_at_apis fallocate)
    string(TOUPPER "${feature}" upper)
    set(check_file "has_${feature}.cpp")
    if(feature STREQUAL "init_priority")
        set(check_file "has_attribute_init_priority.cpp")
    endif()
    check_cxx_source_compiles("#include <${BOOST_SOURCE}/libs/filesystem/config/${check_file}>" BOOST_FILESYSTEM_HAS_${upper})
    if(BOOST_FILESYSTEM_HAS_${upper})
        target_compile_definitions(boost_filesystem PRIVATE BOOST_FILESYSTEM_HAS_${upper})
    endif()
endforeach()
unset(CMAKE_REQUIRED_INCLUDES)
target_compile_definitions(boost_filesystem PRIVATE BOOST_FILESYSTEM_NO_CXX20_ATOMIC_REF BOOST_FILESYSTEM_DISABLE_STATX BOOST_FILESYSTEM_DISABLE_COPY_FILE_RANGE BOOST_FILESYSTEM_DISABLE_GETRANDOM BOOST_FILESYSTEM_DISABLE_ARC4RANDOM)
target_link_libraries(boost_filesystem PUBLIC boost_atomic)
target_link_libraries(boost_thread PUBLIC boost_atomic boost_chrono)
target_link_libraries(boost_coroutine PUBLIC boost_context)
target_link_libraries(boost_process PUBLIC boost_filesystem)

foreach(component data uc i18n)
    add_library(icu_${component} STATIC IMPORTED)
    set_target_properties(icu_${component} PROPERTIES IMPORTED_LOCATION "${ICU_PREFIX}/lib/libicu${component}.a" INTERFACE_INCLUDE_DIRECTORIES "${ICU_PREFIX}/include" INTERFACE_COMPILE_DEFINITIONS U_STATIC_IMPLEMENTATION)
endforeach()
set_target_properties(icu_uc PROPERTIES INTERFACE_LINK_LIBRARIES "icu_data;dl;m")
set_target_properties(icu_i18n PROPERTIES INTERFACE_LINK_LIBRARIES icu_uc)
target_compile_definitions(boost_locale PRIVATE BOOST_LOCALE_WITH_ICU=1 BOOST_LOCALE_NO_STD_BACKEND=1 BOOST_LOCALE_NO_POSIX_BACKEND=1 BOOST_LOCALE_NO_WINAPI_BACKEND=1)
target_link_libraries(boost_locale PUBLIC icu_i18n boost_charconv boost_thread)

string(JSON count LENGTH "${plan}" bzip2_sources)
math(EXPR last "${count}-1")
set(sources)
foreach(index RANGE ${last})
    string(JSON name GET "${plan}" bzip2_sources ${index})
    list(APPEND sources "${BZIP2_SOURCE}/${name}")
endforeach()
add_library(bz2 STATIC ${sources})
target_include_directories(bz2 PUBLIC "${BZIP2_SOURCE}")
target_compile_definitions(bz2 PRIVATE _FILE_OFFSET_BITS=64)
target_link_libraries(boost_iostreams PUBLIC bz2 z boost_random)
install(TARGETS ${install_targets} bz2 ARCHIVE DESTINATION lib)
install(DIRECTORY "${BOOST_SOURCE}/boost" DESTINATION include)
install(FILES "${BZIP2_SOURCE}/bzlib.h" DESTINATION include)
install(FILES "${BOOST_SOURCE}/LICENSE_1_0.txt" DESTINATION share/licenses/wesnoth-boost)
install(FILES "${BZIP2_SOURCE}/LICENSE" DESTINATION share/licenses/wesnoth-boost RENAME bzip2-LICENSE)

add_executable(wesnoth-boost-check probe.cpp)
target_compile_options(wesnoth-boost-check PRIVATE -Wall -Wextra -Werror -fexceptions)
target_link_options(wesnoth-boost-check PRIVATE -pie "-Wl,-Map,${CMAKE_BINARY_DIR}/probe.map")
target_link_libraries(wesnoth-boost-check PRIVATE ${install_targets} bz2 icu_i18n z dl m)
set_target_properties(wesnoth-boost-check PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS wesnoth-boost-check RUNTIME DESTINATION bin)
