cmake_minimum_required(VERSION 3.22)
project(wesnoth_alliance_transport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH ON)

# Explicit existing headers/prefixes; this adapter never downloads dependencies.
set(BOOST_INCLUDE_DIR "" CACHE PATH "Existing Boost include directory")
if(NOT EXISTS "${BOOST_INCLUDE_DIR}/boost/version.hpp")
    message(FATAL_ERROR "BOOST_INCLUDE_DIR must contain the existing Boost headers")
endif()
if(ANDROID)
    if(NOT ANDROID_ABI STREQUAL "x86_64" OR NOT ANDROID_PLATFORM_LEVEL STREQUAL "23" OR
       NOT ANDROID_STL STREQUAL "c++_shared")
        message(FATAL_ERROR "Use Android x86_64/API23/shared-C++ toolchain")
    endif()
    set(NETWORK_PREFIX "" CACHE PATH "Completed Android Network prefix")
    foreach(part ssl crypto)
        if(NOT EXISTS "${NETWORK_PREFIX}/lib/lib${part}.a")
            message(FATAL_ERROR "Completed Android Network prefix is required")
        endif()
        add_library(alliance_${part} STATIC IMPORTED)
        set_target_properties(alliance_${part} PROPERTIES
            IMPORTED_LOCATION "${NETWORK_PREFIX}/lib/lib${part}.a"
            INTERFACE_INCLUDE_DIRECTORIES "${NETWORK_PREFIX}/include")
    endforeach()
    set(alliance_ssl_libraries alliance_ssl alliance_crypto dl)
else()
    find_package(OpenSSL 3.0 REQUIRED COMPONENTS SSL Crypto)
    set(alliance_ssl_libraries OpenSSL::SSL OpenSSL::Crypto)
endif()
find_package(Threads REQUIRED)

add_library(alliance_transport STATIC ../../alliance_native/transport.cpp)
target_include_directories(alliance_transport PUBLIC ../../alliance_native)
target_include_directories(alliance_transport SYSTEM PUBLIC "${BOOST_INCLUDE_DIR}")
target_compile_definitions(alliance_transport PUBLIC BOOST_ALL_NO_LIB BOOST_ERROR_CODE_HEADER_ONLY)
target_compile_options(alliance_transport PRIVATE -Wall -Wextra -Werror)
target_link_libraries(alliance_transport PUBLIC ${alliance_ssl_libraries} Threads::Threads)
if(WIN32)
    target_compile_definitions(alliance_transport PUBLIC _WIN32_WINNT=0x0A00 NOMINMAX)
    target_link_libraries(alliance_transport PUBLIC ws2_32 mswsock)
endif()

add_executable(alliance-native-probe probe.cpp)
target_compile_options(alliance-native-probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(alliance-native-probe PRIVATE alliance_transport)
set_target_properties(alliance-native-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
install(TARGETS alliance-native-probe RUNTIME DESTINATION bin)
