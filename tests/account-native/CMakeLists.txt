cmake_minimum_required(VERSION 3.22)
project(alliance_account_host_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL 3.0 REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(CLIENT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BOOST_INCLUDE_DIR "${CLIENT}/build/android-boost/prefix/include" CACHE PATH "Pinned Boost headers")
add_library(account_client STATIC
    "${CLIENT}/alliance_native/transport.cpp"
    "${CLIENT}/alliance_native/account_protocol.cpp"
    "${CLIENT}/alliance_native/account_client.cpp")
target_include_directories(account_client PUBLIC "${CLIENT}/alliance_native")
target_include_directories(account_client SYSTEM PUBLIC "${BOOST_INCLUDE_DIR}")
target_compile_definitions(account_client PUBLIC BOOST_ALL_NO_LIB BOOST_ERROR_CODE_HEADER_ONLY)
target_compile_options(account_client PRIVATE -Wall -Wextra -Werror)
target_link_libraries(account_client PUBLIC OpenSSL::SSL OpenSSL::Crypto ZLIB::ZLIB Threads::Threads)
foreach(name protocol_check flow_probe response_probe)
    add_executable(account_${name} "${CLIENT}/tests/account_${name}.cpp")
    target_compile_options(account_${name} PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(account_${name} PRIVATE account_client)
endforeach()
add_executable(team_flow_probe "${CLIENT}/tests/team_flow_probe.cpp")
target_compile_options(team_flow_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(team_flow_probe PRIVATE account_client)
enable_testing()
add_test(NAME account_protocol COMMAND account_protocol_check)
